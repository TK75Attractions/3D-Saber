cmake_minimum_required(VERSION 3.16)
project(phonesaber_core LANGUAGES CXX)
add_library(phonesaber_core STATIC src/detection.cpp src/pipeline.cpp src/frame_processor.cpp)
set_target_properties(phonesaber_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(phonesaber_core PUBLIC cxx_std_17)
target_include_directories(phonesaber_core PUBLIC include)
# Swift does not contract multiply/add. Never use fast-math or reassociate sums.
target_compile_options(phonesaber_core PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
option(PHONESABER_BUILD_TOOLS "Build Mac parity tools (not required by JNI)" OFF)
if(PHONESABER_BUILD_TOOLS)
    find_package(ZLIB REQUIRED)
    add_executable(phonesaber_png tools/png_cli.cpp tools/png.cpp)
    target_link_libraries(phonesaber_png PRIVATE phonesaber_core ZLIB::ZLIB)
    target_compile_options(phonesaber_png PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
